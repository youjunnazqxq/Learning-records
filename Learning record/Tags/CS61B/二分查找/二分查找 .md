public class ex06_binarySearch {  
    //实现的前提是数组是按序排序的  
    public  int binarySearch(int [] array, int x){  
        if (array==null||array.length==0){  
            return -1;  
        }  
        return help(array,x,0,array.length);  
    }  
    public int help(int [] array, int x,int left,int right){  
        if(left>right){  
            return -1;  
        }  
        int mid=(left+right)/2；  
        if(array[mid]==x){  
            return mid;  
        }else if(array[mid]<x){  
            return help(array,x,mid+1,right);  
        }else{  
            return help(array,x,left,mid-1);  
        }  
    }  
}
