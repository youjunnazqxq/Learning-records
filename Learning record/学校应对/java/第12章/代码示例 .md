import java.sql.*; // 1. 必须导入 sql 包

public class JdbcQueryDemo {
    public static void main(String[] args) {
        // === 配置数据库信息 (根据你自己的情况修改) ===
        // URL 格式: jdbc:mysql://IP地址:端口/数据库名?参数
        String url = "jdbc:mysql://localhost:3306/school_db?useSSL=false&serverTimezone=UTC";
        String username = "root";   // 你的数据库账号
        String password = "123456"; // 你的数据库密码

        // === 定义 JDBC 对象 (放在 try 外面是为了在 finally 里关闭) ===
        Connection conn = null;
        Statement stmt = null;
        ResultSet rs = null;

        try {
            // 2. 加载驱动 (注册司机)
            // 注意：MySQL 8.0+ 用 cj.jdbc.Driver，老版本用 com.mysql.jdbc.Driver
            Class.forName("com.mysql.cj.jdbc.Driver");

            // 3. 建立连接 (修路) -> 对应 Connection
            System.out.println("正在连接数据库...");
            conn = DriverManager.getConnection(url, username, password);
            System.out.println("连接成功！");

            // 4. 创建执行对象 (买车) -> 对应 Statement
            stmt = conn.createStatement();

            // 5. 执行 SQL 查询 (拉货) -> 对应 ResultSet
            String sql = "SELECT id, name, age FROM student"; // 假设你的表叫 student
            rs = stmt.executeQuery(sql);

            // 6. 处理结果集 (卸货)
            System.out.println("=== 查询结果如下 ===");
            while (rs.next()) {
                // 根据列名或者列的索引(从1开始)取值
                int id = rs.getInt("id");
                String name = rs.getString("name");
                int age = rs.getInt("age");

                System.out.println("学号: " + id + ", 姓名: " + name + ", 年龄: " + age);
            }

        } catch (ClassNotFoundException e) {
            System.out.println("找不到数据库驱动 jar 包！");
            e.printStackTrace();
        } catch (SQLException e) {
            System.out.println("SQL 执行出错！");
            e.printStackTrace();
        } finally {
            // 7. 释放资源 (关门) -> 顺序：后打开的先关闭
            // 这一步非常繁琐，但为了不内存泄漏必须写
            try {
                if (rs != null) rs.close();
                if (stmt != null) stmt.close();
                if (conn != null) conn.close();
                System.out.println("资源已释放");
            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }
}
