# 拖放算法 

1. 在 `mousedown` 上 —— 根据需要准备要移动的元素（也许创建一个它的副本，向其中添加一个类或其他任何东西）。
2. 然后在 `mousemove` 上，通过更改 `position:absolute` 情况下的 `left/top` 来移动它。
3. 在 `mouseup` 上 —— 执行与完成的拖放相关的所有行为。

相关实现：
```
ball.onmousedown = function(event){
//准备阶段 

// 确保绝对定位 & 层级置顶
ball.style.position = 'absolute';
ball.style.zIndex = 1000;
// 移动到body下，使定位相对整个文档
document.body.append(ball);
// 定义“让球中心与鼠标对齐”的函数
function moveAt(pageX, pageY) {
  ball.style.left = pageX - ball.offsetWidth / 2 + 'px';
  ball.style.top = pageY - ball.offsetHeight / 2 + 'px';
}
// 初始化球的位置（鼠标按下时的坐标）
moveAt(event.pageX, event.pageY);


鼠标的移动逻辑


function onMouseMove(event) {
  moveAt(event.pageX, event.pageY);
}
// 绑定鼠标移动事件，实现拖拽跟随
document.addEventListener('mousemove', onMouseMove);



鼠标松开后的清理 
ball.onmouseup = function() {
  // 移除鼠标移动的监听器，停止拖拽
  document.removeEventListener('mousemove', onMouseMove);
  // 清理自身事件，避免重复触发
  ball.onmouseup = null;
};

}

```

默认拖放行为的禁用：

```
ball.ondragstart = function(){

	return false;
}
```

# 修正定位 

在上诉例子中，我们用如下代码来定位，使球的中心位于鼠标指针下方
```
ball.style.left = pageX - ball.foosetWidth / 2 + 'px';

```

果从球的边缘“抓住”球，那么球会突然“跳转”以使球的中心位于鼠标指针下方。

如果我们能够保持元素相对于鼠标指针的初始偏移，那就更好了
当访问者按下按钮（`mousedown`）时 —— 我们可以在变量 `shiftX/shiftY` 中记住鼠标指针到球左上角的距离。我们应该在拖动时保持这个距离。

更新算法：

1.当访问者按下按钮（`mousedown`）时 —— 我们可以在变量 `shiftX/shiftY` 中记住鼠标指针到球左上角的距离。我们应该在拖动时保持这个距离。


```
let shiftX = event.clientX - ball.getBoudningClientRect().left;

let shiftY = event.clientY - ball.getBoundingClientRect().top;
```

2. 然后，在拖动球时，我们将鼠标指针相对于球的这个偏移也考虑在内，像这样
```
ball,style.left = event.pageX - shiftX + 'px';
ball.style.top = eventt.pageY - shiftY + 'px';
```
理解： 球的坐标= 鼠标的坐标再减去偏移量


完整代码：

```
ball.onmousedown = function(event) {
  let shiftX = event.clientX - ball.getBoundingClientRect().left;
  let shiftY = event.clientY - ball.getBoundingClientRect().top;

  ball.style.position = 'absolute';
  ball.style.zIndex = 1000;
  document.body.append(ball);

  moveAt(event.pageX, event.pageY);

  function moveAt(pageX, pageY) {
    ball.style.left = pageX - shiftX + 'px';
    ball.style.top = pageY - shiftY + 'px';
  }

  function onMouseMove(event) {
    moveAt(event.pageX, event.pageY);
  }

  document.addEventListener('mousemove', onMouseMove);

  ball.onmouseup = function() {
    document.removeEventListener('mousemove', onMouseMove);
    ball.onmouseup = null;
  };
};

ball.ondragstart = function() {
  return false;
};

```


# 潜在的放置目标 

有一个叫做 `document.elementFromPoint(clientX, clientY)` 的方法。它会返回在给定的窗口相对坐标处的嵌套的最深的元素（如果给定的坐标在窗口外，则返回 `null`）。如果同一坐标上有多个重叠的元素，则返回最上面的元素。

为什么要采用这个方法呢，因为如果我们不找到这个根部的元素，那么我们拖拽的元素一直是最上层的，他会一直绑定最上层的元素。

方法：
```
ball.hidden = treu;//隐藏我们拖动的元素，避免拖拽

let elemBelow = ducument.elementFromPoint(event.clientX,event.clientY);

ball.hidden = false;
```


## 代码 
```
let currentDroppable = null;

function onMouseMove(event) {
  moveAt(event.pageX, event.pageY);

  ball.hidden = true;
  let elemBelow = document.elementFromPoint(event.clientX, event.clientY);
  ball.hidden = false;

  // mousemove 事件可能会在窗口外被触发（当球被拖出屏幕时）
  // 如果 clientX/clientY 在窗口外，那么 elementFromPoint 会返回 null
  if (!elemBelow) return;

  // 潜在的 droppable 的元素被使用 "droppable" 类进行标记（也可以是其他逻辑）
  let droppableBelow = elemBelow.closest('.droppable');

  if (currentDroppable != droppableBelow) {
    // 我们正在飞入或飞出...
    // 注意：它们两个的值都可能为 null
    //   currentDroppable=null — 如果我们在此事件之前，鼠标指针不是在一个 droppable 上
    //   droppableBelow=null — 如果现在，在当前事件中，我们的鼠标指针不是在一个 droppable 上

    if (currentDroppable) {
      // 处理“飞出” droppable 的元素时的处理逻辑（移除高亮）
      leaveDroppable(currentDroppable);
    }
    currentDroppable = droppableBelow;
    if (currentDroppable) {
      // 处理“飞入” droppable 的元素时的逻辑
      enterDroppable(currentDroppable);
    }
  }
}

```
