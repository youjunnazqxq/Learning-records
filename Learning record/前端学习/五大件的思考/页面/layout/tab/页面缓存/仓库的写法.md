import { defineStore } from "pinia";
import { ref } from "vue";

export const useKeepAliveStore = defineStore("YJ-keepAlive", () => {
  const keepAliveName = ref<string[]>([]);

  const addKeepAlive = (name: string) => {
    !keepAliveName.value.includes(name) && keepAliveName.value.push(name);
  };

  const removeKeepAlive = (name: string) => {
    keepAliveName.value = keepAliveName.value.filter(item => item !== name);
  };

  const setKeepAlive = (name: string[] = []) => {
    keepAliveName.value = name;
  };

  return {
    keepAliveName,
    addKeepAlive,
    removeKeepAlive,
    setKeepAlive
  };
});
