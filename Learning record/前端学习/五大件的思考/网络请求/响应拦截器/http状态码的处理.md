import { ElMessage } from "element-plus";

/**
 * @description: 校验接口响应状态码
 * @param {Number} status
 * @param {String} msg
 * @return void
 */
export const checkStatus = (status: number, msg?: string): void => {
  // const userStore = useUserStore();

  switch (status) {
    case 401:
      ElMessage.error(msg || "登录状态已过期，请重新登录");
      // userStore.setToken("");
      // window.location.href = "/login";
      break;
    case 403:
      ElMessage.error("拒绝访问：您没有权限");
      break;
    case 404:
      ElMessage.error("资源未找到 (404)");
      break;
    case 500:
      ElMessage.error("服务器错误 (500)");
      break;
    case 502:
      ElMessage.error("网关错误 (502)");
      break;
    case 503:
      ElMessage.error("服务不可用 (503)");
      break;
    case 504:
      ElMessage.error("网关超时 (504)");
      break;
    default:
      ElMessage.error(msg || "网络请求错误");
  }
};


HTTP 状态码就像是前端和后端之间沟通的“暗号”。通常，**4 开头**的错误代表**客户端（前端）**的问题，而 **5 开头**的错误代表**服务器（后端）**的问题。

以下是你代码中涉及的各个状态码的具体含义：

### 4xx 客户端错误

这类错误通常意味着前端发出的请求存在问题，比如身份不对、权限不够或者地址写错了。

- **401 (Unauthorized - 未授权)**
    
    - **含义**：**“你是谁？”** 意思是当前请求缺乏有效的身份验证凭证。
        
    - **常见场景**：用户根本没有登录，或者之前登录颁发的 Token 已经过期失效。
        
    - **处理逻辑**：正如你代码注释里写的，通常需要清除本地失效的 Token，并强制跳转回登录页面。
        
- **403 (Forbidden - 禁止访问)**
    
    - **含义**：**“我知道你是谁，但你不能进。”** 意思是服务器理解了请求，也知道你的身份，但是拒绝执行。
        
    - **常见场景**：你是一个普通用户，却尝试调用只有超级管理员才能调用的删除接口。
        
    - **处理逻辑**：提示用户“无权限”，或者将其引导至 403 无权限提示页。
        
- **404 (Not Found - 未找到)**
    
    - **含义**：**“你找的东西不存在。”**
        
    - **常见场景**：前端请求的 URL 路径写错了（多敲了一个字母），或者后端把那个接口/资源删除了。
        
    - **处理逻辑**：检查接口地址是否拼写正确，或与后端确认接口是否已上线。
        

---

### 5xx 服务器错误

这类错误通常意味着前端发送的请求格式是完全正确的，但是后端在处理这个请求时自己“翻车”了。

- **500 (Internal Server Error - 内部服务器错误)**
    
    - **含义**：**“后端代码报错了。”** 这是一个非常通用的服务器错误提示。
        
    - **常见场景**：后端代码里存在空指针异常、除以零错误、或者数据库查询失败等，导致后端程序崩溃，无法正常返回数据。
        
    - **处理逻辑**：前端只能提示服务器错误，需要后端开发人员去查看服务器日志排查 Bug。
        
- **502 (Bad Gateway - 网关错误)**
    
    - **含义**：**“中间人搞砸了。”** * **常见场景**：现在很多项目部署都会用 Nginx 作为网关进行反向代理。502 通常意味着 Nginx 正常工作，但它去向真实的后端服务要数据时，发现后端服务挂掉了（比如 Node/Java 进程死了），Nginx 拿不到数据，只能给你返回 502。
        
- **503 (Service Unavailable - 服务不可用)**
    
    - **含义**：**“服务器现在很忙或正在停机维护。”**
        
    - **常见场景**：双十一并发量太大，服务器资源耗尽撑不住了；或者后端正好在停机发布新版本。这通常是一个临时的状态。
        
- **504 (Gateway Timeout - 网关超时)**
    
    - **含义**：**“后端处理太慢，网关等不及了。”**
        
    - **常见场景**：请求了一个需要处理海量数据的接口，或者后端查询数据库死锁了。网关（如 Nginx）等了很长时间（比如超过 60 秒）都没等到后端的回复，就会直接断开连接并返回 504 给前端。
