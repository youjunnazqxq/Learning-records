public class BST<T extends Comparable<T>> {  
    private Node root;  
    private class  Node{  
        T value;  
        Node left;  
        Node right;  
        public Node(T value){  
            this.value=value;  
            left=null;  
            right=null;  
        }  
    }  
    public BST(T key){  
        root =  new Node (key);  
    }  
    public void delete(T value){  
        this.root=delete_help(this.root,value);  
    }  
    private Node delete_help(Node current_root,T value){  
        if(current_root==null){  
            return null;  
        }  
        int cmp=current_root.value.compareTo(value);  
        if (cmp>0){  
            current_root.left=delete_help(current_root.left,value);  
        }else if(cmp<0){  
            current_root.right=delete_help(current_root.right,value);  
        }else{  
            if(current_root.left==null){  
                return current_root.right;  
            }  
            if(current_root.right==null){  
                return current_root.left;  
            }  
            Node successor=findMin(current_root);  
            current_root.value=successor.value;  
            current_root.right=delete_help(current_root.right,successor.value);  
        }  
        return current_root;  
    }  
    private Node findMin(Node node){  
        while (node.left!=null){  
            node=node.left  
        }  
        return node;  
    }  
    public void insert(T value){  
        this.root=insert_help(this.root,value);  
    }  
    private Node insert_help(Node current_Node,T value){  
        if (current_Node==null){  
            Node newNode= new Node(value);  
            return newNode;  
        }  
        int cmp=current_Node.value.compareTo(value);  
        if(cmp==0){  
            return current_Node;  
        }else if(cmp >0){  
             current_Node.left=insert_help(current_Node.left,value);  
        }else{  
             current_Node.right=insert_help(current_Node.right,value);  
        }  
        return current_Node;  
    }  
    public Node find(T value){  
       return find_help(this.root,value);  
    }  
    private Node find_help(Node current_Node,T value){  
        if(current_Node==null){  
            return null;  
        }  
        int cmp=current_Node.value.compareTo(value);  
        if(cmp==0){  
            return current_Node;  
        }else if(cmp>0){  
            return find_help(current_Node.left,value);  
        }else{  
            return find_help(current_Node.right,value);  
        }  
    }  
}
