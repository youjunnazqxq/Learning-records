<%*
// =====================================================
// 周模板：文件名形如 2025-W01
// today = 该周周一（ISO week 的第一天）
// =====================================================
// ---------- 0) 强制英文 locale（锁定） ----------
window.moment.locale("en");   
// ---------- 1) 从文件名解析 year + week ----------
let title = tp.file.title;                       // e.g. 2025-W01
let m = title.match(/^(\d{4})-W(\d{2})$/);

if (!m) {
  new Notice("⚠️ 文件名不符合 YYYY-W01 格式，例如 2025-W01");
}

let year = m ? m[1] : tp.date.now("GGGG");        // ISO week-year fallback
let week = m ? parseInt(m[2], 10) : parseInt(tp.date.now("WW"), 10);

// ---------- 2) 算出该周周一（ISO 周） ----------
let weekStart = window.moment().isoWeekYear(parseInt(year, 10)).isoWeek(week).startOf("isoWeek"); // Monday
let weekEnd   = weekStart.clone().endOf("isoWeek"); // Sunday

// ---------- 3) 你原来用到的变量（全部改为基于 weekStart） ----------
let today     = weekStart.format("YYYY-MM-DD");    // ✅ 本周第一天（周一）
let toyear    = weekStart.format("YYYY");
let tomonth   = weekStart.format("MMMM");          // 这个周的周一所在月份
let month_path = `${weekStart.format("YYYY")}-M${weekStart.format("MM")}`;
// ---------- 2.1 单独提取周标识 ----------
let weekNum = String(week).padStart(2, "0"); // 01
let weekTag = `W${weekNum}`;                 // W01

// 你原来把 inputDate 作为“今日”，现在固定用周一
let inputDate = today;

// ---------- 4) 标题/前后一天（如果你周模板里还需要“某一天”的导航） ----------
let titleName   = weekStart.format("MM-DD_ddd");   // 周一的 MM-DD_ddd
let before_date = weekStart.clone().add(-1, "days").format("YYYY-MM-DD_ddd"); // 上周日
let after_date  = weekStart.clone().add(1, "days").format("YYYY-MM-DD_ddd");  // 周二

// ---------- 5) 文件元信息（保留） ----------
let createTime = tp.file.creation_date().format("MM-DD");
let modificationDate = tp.file.last_modified_date("dddd Do MMMM YYYY HH:mm:ss");

// ---------- 6) 近七天：生成这一周周一~周日 ----------
let arr = [];
for (let i = 0; i < 7; i++) {
  arr[i] = weekStart.clone().add(i, "days").format("YYYY-MM-DD");
}

// 如果你一定要 yyyy（不建议），这里改成 YYYY 更标准
for (let i = 0; i < 7; i++) {
  arr[i] = window.moment(arr[i], "YYYY-MM-DD").format("YYYY-MM-DD");
}
-%>

tp.file.cursor()
